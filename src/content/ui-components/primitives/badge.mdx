## Install

You can add the primitive via Tiptap CLI

```bash
npx @tiptap/cli@latest add badge
```

## Usage

```tsx
import { Badge } from '@/components/tiptap-ui-primitive/badge'

export default function MyComponent() {
  return (
    <Badge variant="gray">
      <span className="tiptap-badge-text">Resolved</span>
    </Badge>
  )
}
```

## Props

### Badge

| Name | Values | Default |
| --- | --- | --- |
| `variant` | `ghost`, `white`, `gray`, `green`, `yellow`, `default` | Unset |
| `size` | `small`, `compact`, `default`, `large` | `default` |
| `appearance` | `default`, `subdued`, `emphasized` | `default` |
| `trimText` | `boolean` | `false` |
